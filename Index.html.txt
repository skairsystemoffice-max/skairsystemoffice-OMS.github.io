<script>
    // Predefined rates (same as dropdown)
    const rates = {
        '1': 2.50,
        '2': 4.00,
        '3': 5.50,
        '4': 7.00,
        '5': 8.50
    };
    const GST_RATE = 0.18;

    // DOM elements
    const thicknessSelect = document.getElementById('thickness');
    const lengthInput = document.getElementById('length');
    const widthInput = document.getElementById('width');
    const quantityInput = document.getElementById('quantity');
    const gstRadios = document.querySelectorAll('input[name="gst"]');
    const selectedThicknessSpan = document.getElementById('selectedThickness');
    const rateDisplay = document.getElementById('rateDisplay');
    const areaDisplay = document.getElementById('areaDisplay');
    const subtotalDisplay = document.getElementById('subtotalDisplay');
    const gstDisplay = document.getElementById('gstDisplay');
    const totalDisplay = document.getElementById('totalDisplay');
    const gstRow = document.getElementById('gstRow');
    const submitBtn = document.getElementById('submitBtn');
    const resetBtn = document.getElementById('resetBtn');
    const orderListDiv = document.getElementById('orderList');

    // Load orders from localStorage
    let orders = JSON.parse(localStorage.getItem('steelSheetOrders')) || [];

    // Update calculation on any input change
    function updateCalculation() {
        const thickness = thicknessSelect.value;
        const length = parseFloat(lengthInput.value) || 0;
        const width = parseFloat(widthInput.value) || 0;
        const quantity = parseInt(quantityInput.value) || 0;
        const gstSelected = document.querySelector('input[name="gst"]:checked').value === 'yes';

        // Update price info
        if (thickness && rates[thickness]) {
            selectedThicknessSpan.textContent = thickness + 'mm';
            rateDisplay.textContent = '$' + rates[thickness].toFixed(2);
        } else {
            selectedThicknessSpan.textContent = '—';
            rateDisplay.textContent = '$0.00';
        }

        // Calculate
        const pricePerSqft = rates[thickness] || 0;
        const area = length * width;
        const subtotal = area * pricePerSqft * quantity;
        const gstAmount = gstSelected ? subtotal * GST_RATE : 0;
        const total = subtotal + gstAmount;

        // Display
        areaDisplay.textContent = area.toFixed(2) + ' sqft';
        subtotalDisplay.textContent = '$' + subtotal.toFixed(2);
        gstDisplay.textContent = '$' + gstAmount.toFixed(2);
        totalDisplay.textContent = '$' + total.toFixed(2);

        if (gstSelected) {
            gstRow.style.display = 'flex';
        } else {
            gstRow.style.display = 'none';
        }
    }

    // Attach event listeners
    thicknessSelect.addEventListener('change', updateCalculation);
    lengthInput.addEventListener('input', updateCalculation);
    widthInput.addEventListener('input', updateCalculation);
    quantityInput.addEventListener('input', updateCalculation);
    gstRadios.forEach(radio => radio.addEventListener('change', updateCalculation));

    // Initial calculation
    updateCalculation();

    // Submit order
    document.getElementById('quoteForm').addEventListener('submit', function(e) {
        e.preventDefault();

        const thickness = thicknessSelect.value;
        if (!thickness) {
            alert('Please select a sheet thickness.');
            return;
        }
        const length = parseFloat(lengthInput.value);
        const width = parseFloat(widthInput.value);
        const quantity = parseInt(quantityInput.value);
        if (!length || !width || !quantity) {
            alert('Please fill in all dimensions and quantity.');
            return;
        }

        const gstSelected = document.querySelector('input[name="gst"]:checked').value === 'yes';
        const pricePerSqft = rates[thickness];
        const area = length * width;
        const subtotal = area * pricePerSqft * quantity;
        const gstAmount = gstSelected ? subtotal * GST_RATE : 0;
        const total = subtotal + gstAmount;

        const order = {
            id: Date.now(),
            thickness: thickness + 'mm',
            length: length,
            width: width,
            quantity: quantity,
            area: area,
            pricePerSqft: pricePerSqft,
            gst: gstSelected,
            subtotal: subtotal,
            gstAmount: gstAmount,
            total: total,
            createdAt: new Date().toLocaleString()
        };

        orders.push(order);
        localStorage.setItem('steelSheetOrders', JSON.stringify(orders));
        renderOrders();
        resetForm();
        alert('✅ Order submitted successfully!');
    });

    // Reset form
    resetBtn.addEventListener('click', resetForm);
    function resetForm() {
        thicknessSelect.value = '';
        lengthInput.value = '';
        widthInput.value = '';
        quantityInput.value = '1';
        document.querySelector('input[name="gst"][value="no"]').checked = true;
        updateCalculation();
    }

    // Render order history
    function renderOrders() {
        if (orders.length === 0) {
            orderListDiv.innerHTML = '<div class="empty-orders">No orders yet. Submit your first quote above.</div>';
            return;
        }

        let html = '';
        orders.forEach((order, index) => {
            const gstLabel = order.gst ? 'With GST (18%)' : 'Non-GST';
            html += `
                <div class="order-card">
                    <div class="order-details">
                        <strong>Order #${index + 1}</strong> (${order.createdAt})<br>
                        Thickness: ${order.thickness} | ${order.length}ft × ${order.width}ft × ${order.quantity} sheets<br>
                        Area: ${order.area.toFixed(2)} sqft | Rate: $${order.pricePerSqft.toFixed(2)}/sqft<br>
                        Subtotal: $${order.subtotal.toFixed(2)} | GST: $${order.gstAmount.toFixed(2)} | <strong>Total: $${order.total.toFixed(2)}</strong><br>
                        ${gstLabel}
                    </div>
                    <div class="order-actions">
                        <button class="edit-btn" onclick="editOrder(${order.id})">✏️ Edit</button>
                        <button class="delete-btn" onclick="deleteOrder(${order.id})">🗑️ Delete</button>
                    </div>
                </div>
            `;
        });
        orderListDiv.innerHTML = html;
    }

    // Edit order: load data into form and remove from list
    window.editOrder = function(id) {
        const order = orders.find(o => o.id === id);
        if (!order) return;

        // Fill form
        thicknessSelect.value = order.thickness.replace('mm', '');
        lengthInput.value = order.length;
        widthInput.value = order.width;
        quantityInput.value = order.quantity;
        if (order.gst) {
            document.querySelector('input[name="gst"][value="yes"]').checked = true;
        } else {
            document.querySelector('input[name="gst"][value="no"]').checked = true;
        }
        updateCalculation();

        // Remove from orders
        orders = orders.filter(o => o.id !== id);
        localStorage.setItem('steelSheetOrders', JSON.stringify(orders));
        renderOrders();

        // Scroll to top
        window.scrollTo({ top: 0, behavior: 'smooth' });
    };

    // Delete order
    window.deleteOrder = function(id) {
        if (!confirm('Delete this order?')) return;
        orders = orders.filter(o => o.id !== id);
        localStorage.setItem('steelSheetOrders', JSON.stringify(orders));
        renderOrders();
    };

    // Initial render
    renderOrders();
</script>
